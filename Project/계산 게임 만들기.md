n='n'

    x=int(input('문제 수를 선택하시오:'))

        y=int(input('답:'))

            l+=1
        elif c != y:

            w+=1
            
    print('문제풀이 끝!')
    print('오답수:%d, 정답수:%d'%(w,l))
